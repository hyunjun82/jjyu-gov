 function savePageVisit() {
 var menuUrl = encodeURI(window.location.pathname + window.location.search);
 var menuName = "";
 var pageList = getCookie("recentPages");
 var pages = pageList ? JSON.parse(pageList) : [];
 pages = pages.filter(function(page) {
 return page.menuUrl !== menuUrl;
 });
 pages.unshift({ menuUrl: menuUrl, menuName: menuName });
 if (pages.length > 5) {
 pages = pages.slice(0, 5);
 }
 setCookie("recentPages", JSON.stringify(pages), 1);
 }
 function displayRecentPages() {
 var pageList = getCookie("recentPages");
 var pages = pageList ? JSON.parse(pageList) : [];
 var recentUls = document.getElementsByClassName("recentUl");
 for (var i = 0; i < recentUls.length; i++) {
 var recentUl = recentUls[i];
 recentUl.innerHTML = "";
 if (pages.length === 0) {
 $(".numC").html("0");
 } else {
 var countV = 0;
 pages.forEach(function(page) {
 var li = document.createElement("li");
 li.innerHTML = '<a href="' + decodeURI(page.menuUrl) + '">' + page.menuName + '</a>';
 recentUl.appendChild(li);
 countV++;
 });
 $(".numC").html(countV);
 }
 }
 }
