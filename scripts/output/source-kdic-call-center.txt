 function initMyMenu(){
 var setCnt = 0;
 $("#myMenuArea div a.hp_whp_menu_empty").not(":last").remove();
 $("input[id^=hp_whp_menu]").sort(function (a, b){
 return gfn_OnlyNumber(b.id) - gfn_OnlyNumber(a.id)
 }).each(function(){
 var id = $(this).attr("id");
 if("Y" == gfn_getCookie(myMenuPrefix+id)){
 setCnt += 1;
 $("#myMenuArea div a:first").before($("#myMenuArea div a."+id));
 $("#myMenuEmptyMsg").addClass("dev-hidden");
 $("#"+id).attr("checked", true);
 $("#myMenuArea div a."+id).removeClass("dev-hidden");
 } else {
 $("#"+id).removeAttr("checked");
 $("#myMenuArea div a."+id).addClass("dev-hidden");
 }
 });
 if(setCnt == 0) {
 $("#myMenuEmptyMsg").after($("#myMenuArea div a"));
 $("#myMenuEmptyMsg").removeClass("dev-hidden");
 } else if(setCnt < 9){
 for(var i=0; i<9-setCnt; i++){
 console.log(i);
 var empty = $("#myMenuArea div a.hp_whp_menu_empty:last").clone();
 $(empty).removeClass("dev-hidden");
 $("#myMenuArea div a").not(".dev-hidden").last().after(empty);
 }
 }
 }
