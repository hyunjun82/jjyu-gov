function comma(num){
 var len, point, str; 
 num = num + ""; 
 point = num.length % 3 ;
 len = num.length; 
 str = num.substring(0, point); 
 while (point < len) { 
 if (str != "") str += ","; 
 str += num.substring(point, point + 3); 
 point += 3; 
 } 
 return str;
}
function getWaitTime(session_time_out, limitCnt, waitCnt) {
 if ( limitCnt == 0 ) return '알수없음';
 var aa = session_time_out/limitCnt * waitCnt;
 aa = Math.round(aa);
 var hour='00';
 var min='00'
 var sec='00';
 if ( aa >=3600 ) {
 hour=Math.floor(aa/3600); 
 if ( hour < 10 ) hour = '0'+hour;
 aa=aa-hour*3600;
 } 
 if ( aa >=60 ) {
 min=Math.floor(aa/60);
 if ( min < 10 ) min = '0'+min;
 aa=aa-min*60;
 } 
 if ( aa < 10 ) sec = '0'+aa;
 else sec = aa;
 var result = hour + ":" + min + ":" + sec;
 return result;
}
