안전조사보고서
한국소비자원에서 수행한 안전실태조사 및 위해정보 심층분석 등 보고서를 확인할 수 있습니다.

function comma(num){
 var len, point, str; 
 num = num + ""; 
 point = num.length % 3 ;
 len = num.length; 
 str = num.substring(0, point); 
 while (point < len) { 
 if (str != "") str += ","; 
 str += num.substring(point, point + 3); 
 point += 3; 
 } 
 return str;
}
