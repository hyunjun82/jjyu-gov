 function fn_interestSetSave(){
 var name = "a1.setInterest";
 var codes = new Array(); //샘플 -> [{"parent_cd_arr":["PLCY0001","PLCY0002"]},{"parent_cd":"PLCY0001","child_cd":["PLCY0301","PLCY0401"]},{"parent_cd":"PLCY0002","child_cd":["PLCY0202"]}]
 //부모코드 목록
 var parent_cd_arr = new Array();
 var child_codes = new Array(); 
 var pCd = "";
 $('input[name=keyCodeCd]:checked').each(function(idx){
 var parent_cd = $(this).prev().val();
 var code_cd = $(this).val();
 //부모코드 리스트 만들기
 if(parent_cd != pCd){
 parent_cd_arr.push(parent_cd);
 pCd = parent_cd;
 }
 var codeInfo = new Object();
 codeInfo.parent_cd = parent_cd;
 codeInfo.code_cd = code_cd;
 child_codes.push(codeInfo);
 });
 var pObj = new Object();
 pObj.parent_cd_arr = parent_cd_arr;
 codes.push(pObj);
 for(var i=0; i<parent_cd_arr.length; i++){
 var obj = new Object();
 var parent_cd_str = parent_cd_arr[i]; 
 obj.parent_cd = parent_cd_str;
 var cArr = new Array();
 for(var j=0; j<child_codes.length; j++){
 var c_parent_cd = child_codes[j].parent_cd;
 if(parent_cd_str == c_parent_cd){
 cArr.push(child_codes[j].code_cd);
 }
 }
 obj.child_cd = cArr;
 codes.push(obj);
 }
 var jsonInfo = encodeURI(JSON.stringify(codes));
 var expiredays = 3650; 
 var expire_date = new Date() ; //오늘날짜를 대입값으로 설정한다.
 var secure = location.protocol == 'https:' ? ' secure;' : '';
 expire_date.setDate(expire_date.getDate() + expiredays) ;
 $.ajax({
 url : "/interestSave.es?name="+name+"&codes="+jsonInfo+"&sid=a1&code_tmp="+codes,
 cache: false,
 success:function(result){
 $('#interestList').html(result);
 alert("관심키워드 설정이 완료되었습니다.");
 $('.custom-search-settingBox').removeClass('on');
 },
 error:function(data,status,err){
// console.log(err);
 }
 });
 }

 $('input[name=keyCodeCd]:checked').each(function(){
 $(this).prop("checked",false);
 });
 fn_interestSetSave();
 }
 }
