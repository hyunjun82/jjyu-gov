    function calcHeight(tries) {
      var isVerticalType = Boolean(document.getElementsByClassName('widget--v').length)

      var nameHeight = [];
      var authorHeight = [];
      var priceHeight = 0;
      var origPriceHeight = 0;
      var salePriceHeight = 0;
      var pricePadding = 0;
      var thumbnailWidth = 0;

      var itemList = document.querySelectorAll('.item a');
      for (var i = 0; i < itemList.length; i++) {
        // 'name' height
        if (!isVerticalType) {
          var rowIdx = i % 2;

          if (itemList[i].getElementsByClassName('name').length > 0) {
            nameHeight[rowIdx] = Math.max(nameHeight[rowIdx] ? nameHeight[rowIdx] : 0, itemList[i].getElementsByClassName('name')[0].clientHeight);
            // max-height for .name is 41px
            nameHeight[rowIdx] = (nameHeight[rowIdx] > 41) ? 41 : nameHeight[rowIdx];
          }
          if (itemList[i].getElementsByClassName('author').length > 0) {
            authorHeight[rowIdx] = Math.max(authorHeight[rowIdx] ? authorHeight[rowIdx] : 0, itemList[i].getElementsByClassName('author')[0].clientHeight);
          }
        }

        if (itemList[i].getElementsByClassName('thumbnail').length > 0) {
          // 'thumbnail' width
          thumbnailWidth = Math.max(thumbnailWidth, itemList[i].getElementsByClassName('thumbnail')[0].clientWidth);
        }

        if (itemList[i].getElementsByClassName('price').length > 0) {
          // 'price' height
          var priceElement = itemList[i].getElementsByClassName('price')[0];
          priceHeight = Math.max(priceHeight, priceElement.clientHeight - (parseInt(getComputedStyle(priceElement).paddingTop) + parseInt(getComputedStyle(priceElement).paddingBottom)));
        }

        if (itemList[i].getElementsByClassName('original_price').length > 0) {
          // 'original_price' height
          origPriceHeight = Math.max(origPriceHeight, itemList[i].getElementsByClassName('original_price')[0].clientHeight);
        }

        if (itemList[i].getElementsByClassName('sale_price').length > 0) {
          // 'sale_price' height
          salePriceHeight = Math.max(salePriceHeight, itemList[i].getElementsByClassName('sale_price')[0].clientHeight);
        }
      }

      if (salePriceHeight > 0 && origPriceHeight > 0) {
        pricePadding = Math.floor(Math.min.apply(null, [salePriceHeight, origPriceHeight].filter(Boolean)) * 0.2 * 100) / 100;
      }

      for (var i = 0; i < itemList.length; i++) {
        if (!isVerticalType) {
          if (itemList[i].getElementsByClassName('name').length > 0) {
            itemList[i].getElementsByClassName('name')[0].style.height = String(nameHeight[i % 2]) + 'px';
          }
          if (itemList[i].getElementsByClassName('author').length > 0) {
            itemList[i].getElementsByClassName('author')[0].style.height = String(authorHeight[i % 2]) + 'px';
          }
        }

        if (itemList[i].getElementsByClassName('thumbnail').length > 0) {
          itemList[i].getElementsByClassName('thumbnail')[0].style.width = String(thumbnailWidth) + 'px';
        }

        if (itemList[i].getElementsByClassName('price').length > 0) {
          if (priceHeight > 0) {
            // 'price' height
            itemList[i].getElementsByClassName('price')[0].style.height = String(priceHeight) + 'px';
          }

          if (salePriceHeight > 0 && origPriceHeight > 0) {
            itemList[i].getElementsByClassName('price')[0].style.paddingTop = String(pricePadding) + 'px';
          }
        }

        if (itemList[i].getElementsByClassName('original_price').length > 0 && origPriceHeight > 0) {
          // 'original_price' height
          itemList[i].getElementsByClassName('original_price')[0].style.height = String(origPriceHeight) + 'px';
        }

        if (itemList[i].getElementsByClassName('sale_price').length > 0 && salePriceHeight > 0) {
          // 'sale_price' height
          itemList[i].getElementsByClassName('sale_price')[0].style.height = String(salePriceHeight) + 'px';
        }
      }

      // image load 안된 시점에 onload가 발생시 정상 높이 계산이 되지 않음
      // image height가 0인 경우 10번 retry
      // if ($('img.thumbnail').height() === 0) {
      var imgThumbnail = document.querySelector('img.thumbnail');
      if (true && imgThumbnail.clientHeight === 0) {
        if (tries < 10) {
          setTimeout(function () { calcHeight(tries + 1); }, 100);
        }
      } else {
        syncHeight(100);
      }
    }

    function syncHeight(ms) {
      setTimeout(function() {
        var c = window.getComputedStyle(document.body);
        var margin = parseFloat(c.marginTop) + parseFloat(c.marginBottom)
        var height = document.body.offsetHeight + margin || 1;
        var has_recom = height > 1 ? 1 : 0;

        simplePostMessage.postMessage('id=recopick_widget_ETbRd9iK' + '&if_height=' + height + '&has_recom=' + has_recom, 'https://www.vans.co.kr/');
      }, ms);
    }

    window.onload = function () {
      calcHeight();
    }

    window.onresize = function () {
      syncHeight(100);
    }
