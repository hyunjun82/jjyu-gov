(function () {
  var STORAGE_KEY = "openedSessionStores";
  var CHANNEL = "store-alert-storage";

  function getParentRoot(hostname) {
    var parts = String(hostname || "").split(".").filter(Boolean);
    if (parts.length <= 2) return hostname;
    return parts.slice(-2).join(".");
  }

  var bridgeRoot = getParentRoot(window.location.hostname);

  function isAllowedOrigin(origin) {
    try {
      var host = new URL(origin).hostname;
      var root = getParentRoot(host);
      return root === bridgeRoot || host === window.location.hostname;
    } catch (e) {
      return false;
    }
  }

  function getLocalMidnightTs() {
    var midnight = new Date();
    midnight.setHours(24, 0, 0, 0);
    return midnight.getTime();
  }

  function readRecord() {
    try {
      var raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return { expiresAt: getLocalMidnightTs(), sessions: [] };
      var parsed = JSON.parse(raw);
      if (Array.isArray(parsed)) {
        return { expiresAt: getLocalMidnightTs(), sessions: parsed };
      }
      if (!parsed || typeof parsed !== "object") {
        return { expiresAt: getLocalMidnightTs(), sessions: [] };
      }
      if (!parsed.expiresAt || Date.now() >= parsed.expiresAt) {
        localStorage.removeItem(STORAGE_KEY);
        return { expiresAt: getLocalMidnightTs(), sessions: [] };
      }
      return {
        expiresAt: parsed.expiresAt,
        sessions: Array.isArray(parsed.sessions) ? parsed.sessions : []
      };
    } catch (e) {
      return { expiresAt: getLocalMidnightTs(), sessions: [] };
    }
  }

  function writeRecord(sessions) {
    var record = {
      expiresAt: getLocalMidnightTs(),
      sessions: Array.isArray(sessions) ? sessions : []
    };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(record));
    return record.sessions;
  }

  function reply(source, origin, requestId, ok, data, error) {
    if (!source) return;
    source.postMessage(
      {
        channel: CHANNEL,
        type: "response",
        requestId: requestId,
        ok: !!ok,
        data: data,
        error: error || null
      },
      origin
    );
  }

  window.addEventListener("message", function (event) {
    if (!event || !event.data || event.data.channel !== CHANNEL) return;
    if (!isAllowedOrigin(event.origin)) return;

    var msg = event.data;
    var requestId = msg.requestId;
    var type = msg.type;

    try {
      if (type === "ping") {
        reply(event.source, event.origin, requestId, true, { ready: true });
        return;
      }
      if (type === "get") {
        reply(event.source, event.origin, requestId, true, readRecord().sessions);
        return;
      }
      if (type === "set") {
        reply(event.source, event.origin, requestId, true, writeRecord(msg.sessions));
        return;
      }
      if (type === "remove") {
        var current = readRecord().sessions;
        var next = current.filter(function (s) {
          return s && s.sessionCode !== msg.sessionCode;
        });
        reply(event.source, event.origin, requestId, true, writeRecord(next));
        return;
      }
      if (type === "clear") {
        localStorage.removeItem(STORAGE_KEY);
        reply(event.source, event.origin, requestId, true, []);
        return;
      }
    } catch (err) {
      reply(
        event.source,
        event.origin,
        requestId,
        false,
        null,
        String(err && err.message ? err.message : err)
      );
    }
  });

  try {
    if (window.parent && window.parent !== window) {
      window.parent.postMessage({ channel: CHANNEL, type: "ready" }, "*");
    }
  } catch (e) {}
})();
